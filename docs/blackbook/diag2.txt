=== 2) the interconnect: go.mjs argv contract (already recovered earlier), and does it accept MULTIPLE inputs → a single native-scale pdf of concatenated pages? (that is how a 4-per-page plate could be one pdf) ===
import puppeteer from 'puppeteer';
import fs from 'node:fs';
const [inFile, outFile] = process.argv.slice(2);
const svg = fs.readFileSync(inFile, 'utf8');
const html = `<!doctype html><html><head><style>html,body{margin:0;padding:0}svg{display:block}</style></head><body>${svg}</body></html>`;
const browser = await puppeteer.launch({headless: true, args:['--no-sandbox','--disable-dev-shm-usage']});
const page = await browser.newPage();
await page.setContent(html, {waitUntil:'networkidle0'});
const dim = await page.evaluate(() => { const s=document.querySelector('svg'); return {w:parseFloat(s.getAttribute('width')), h:parseFloat(s.getAttribute('height'))}; });
const inches = (px) => ((px*0.75)/72)+'in';
await page.pdf({path: outFile, width: inches(dim.w), height: inches(dim.h), printBackground:true, margin:{top:'0',right:'0',bottom:'0',left:'0'}});
console.log('ok', dim.w, dim.h);
await browser.close();

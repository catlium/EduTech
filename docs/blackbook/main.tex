% ---------- Headings: 20 pt bold centered ALL CAPS chapter titles ----------
% Chapter opener pages are CENTERED ON BOTH AXES: the title block is
% pushed to the vertical middle of the text block (vfill before+after) and
% horizontally centered inside it. Hand-rolled in the class internals so we
% control vertical placement exactly (titlesec cannot center a display
% block vertically on the page).
\makeatletter
\renewcommand{\@makechapterhead}[1]{%
  \vspace*{\fill}%
  {\parindent \z@ \centering \normalfont
   \large \bfseries \MakeUppercase{\@chapapp} \ \thechapter \par
   \vspace{12pt}%
   \fontsize{20}{24}\selectfont \bfseries #1\par}%
  \vspace*{\fill}\par}
\renewcommand{\@makeschapterhead}[1]{%
  {\parindent \z@ \centering \normalfont
   \fontsize{20}{24}\selectfont \bfseries #1\par}%
  \vspace{40pt}}
\makeatother

}

% ---------- Inserted blank pages carry NO page number ----------
% The kernel \cleardoublepage on this TeX Live does not apply
% \thispagestyle{empty} to the blank it inserts, so the blank would
% inherit the active pagestyle and print a (roman) page number. That
% violates the institutional rule that an inserted blank page must
% contain no page number, header, footer, text, or decoration. Redefine
% it to force the empty pagestyle on the inserted blank.
\makeatletter
\newcommand{\bb@cleardoublepage}{%
  \clearpage
  \if@twoside
    \ifodd\c@page
    \else
      \hbox{}%
      \thispagestyle{empty}%
      \newpage
      \if@twocolumn\hbox{}\newpage\fi
    \fi
  \fi
}
\let\cleardoublepage\bb@cleardoublepage
\makeatother
